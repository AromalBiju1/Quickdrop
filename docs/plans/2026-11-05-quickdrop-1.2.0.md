# QuickDrop 1.2.0 Implementation Plan

> **For the implementing engineer:** execute this plan task by task. Each task ends with a commit. Do not skip the failing-test step for behavior changes. Migration tasks use the existing test suite as their safety net.

**Goal:** Ship QuickDrop 1.2.0 with a theme-aware dark mode, text sending from Android / Linux / Windows and the browser receiver, and subtle motion that respects reduced-motion.

**Architecture:** A `QuickDropPalette` `ThemeExtension` replaces the static color class. Text travels through the existing `POST /api/transfer` endpoint with an `X-Text-Snippet: 1` header, so 1.1.0 receivers still save a normal `.txt`. Motion uses Flutter built-ins behind a reduced-motion helper.

**Tech stack:** Flutter 3.47.7 / Dart, Material 3, `file_picker`, `path_provider`, `desktop_drop`, existing GitHub Actions desktop release workflow. No new packages.

**Spec:** `docs/design/2026-11-04-quickdrop-1.2.0-design.md`

**Step 4: Rebuild `QuickDropTheme` on the palette**

Replace the existing `QuickDropTheme` class with a builder shared by both themes. Keep the same widget-level theme settings as 1.1.0, with these rules:

- `QuickDropTheme.light` calls `_build(QuickDropPalette.light, Brightness.light)`.
- `QuickDropTheme.dark` calls `_build(QuickDropPalette.dark, Brightness.dark)`.
- Remove the `lightTheme` and `darkTheme` aliases.
- Build the scheme with `brightness == Brightness.light ? const ColorScheme.light(...) : const ColorScheme.dark(...)`, using the same `p.*` field list as 1.1.0. Set `ThemeData(brightness: brightness, ...)` as well.
- Register the palette with `extensions: [p]`.
- Use `p.*` for every color. Use `p.onAccent` for `ColorScheme.onPrimary`, `onTertiary`, `onError` and the `FilledButton` foreground. For `ColorScheme.onSecondary`, use `p.onAccent` in light and `p.textPrimary` in dark, because dark `secondary` is a dark surface.

- Use `p.shadowSoft` / `p.shadowRaised` for card, dialog and tooltip shadow colors.
- Use `p.surface` for cards, dialogs, sheets, menus, app bars and popup menus.
- Use `p.background` for the scaffold and `p.surfaceMuted` for input fills.
- Use `p.border` for outlines and `p.surfaceSubtle` for dividers and progress tracks.

```dart
import 'dart:convert';

/// Constants and helpers shared by the text sender, receiver and browser page.
class TextTransfer {
  static const String header = 'X-Text-Snippet';
  static const int maxCharacters = 16000;
  static const int maxBytes = 64 * 1024;

  static String fileName({DateTime? at}) {
    final time = at ?? DateTime.now();
    String two(int value) => value.toString().padLeft(2, '0');
    return 'QuickDrop-text-${time.year}${two(time.month)}${two(time.day)}'
        '-${two(time.hour)}${two(time.minute)}${two(time.second)}.txt';
  }

  static int byteLength(String text) => utf8.encode(text).length;

  /// Returns a user-facing error, or null when the text can be sent.
  static String? validate(String text) {
    if (text.trim().isEmpty) return 'Nothing to send.';
    if (text.length > maxCharacters) {
      return 'Text is limited to $maxCharacters characters.';
    }
    if (byteLength(text) > maxBytes) {
      return 'Text is limited to 64 KiB.';
    }
    return null;
  }

  /// First 500 characters, used by history. Keeps line breaks.
  static String preview(String text, {int limit = 500}) {
    if (text.length <= limit) return text;
    return '${text.substring(0, limit)}…';
  }

  /// One-line, 80-character preview for notifications and banners.
  static String notificationPreview(String text, {int limit = 80}) {
    final flat = text.replaceAll(RegExp(r'\s+'), ' ').trim();
    if (flat.length <= limit) return flat;
    return '${flat.substring(0, limit)}…';
  }
}
```

```dart
Future<void> _handleIncomingText(HttpRequest request) async {
  File? tempFile;
  try {
    final bytes = <int>[];
    var totalBytes = 0;

    // Read the whole body but only keep the first 64 KiB. Draining the rest
    // lets the sender receive the 413 instead of a connection reset.
    await for (final chunk in request) {
      totalBytes += chunk.length;
      if (totalBytes <= TextTransfer.maxBytes) {
        bytes.addAll(chunk);
      }
    }

    if (totalBytes > TextTransfer.maxBytes) {
      await _writeJsonError(
        request,
        HttpStatus.requestEntityTooLarge,
        'Text is limited to 64 KiB.',
      );
      return;
    }

    String text;
    try {
      text = utf8.decode(bytes);
    } on FormatException {
      await _writeJsonError(
        request,
        HttpStatus.badRequest,
        'Text must be UTF-8.',
      );
      return;
    }

    if (text.trim().isEmpty) {
      await _writeJsonError(request, HttpStatus.badRequest, 'Text is empty.');
      return;
    }

    if (text.length > TextTransfer.maxCharacters) {
      await _writeJsonError(
        request,
        HttpStatus.requestEntityTooLarge,
        'Text is limited to ${TextTransfer.maxCharacters} characters.',
      );
      return;
    }

    final rawName = request.headers.value('X-File-Name');
    var fileName = FileUtils.sanitizeFileName(
      rawName != null && rawName.isNotEmpty
          ? Uri.decodeComponent(rawName)
          : TextTransfer.fileName(),
    );
    if (!fileName.toLowerCase().endsWith('.txt')) {
      fileName = '$fileName.txt';
    }
    final senderName = Uri.decodeComponent(
      request.headers.value('X-Sender-Name') ?? 'Unknown',
    );

    final saveDir = await getSaveDirectory(fileName: fileName);
    final targetFile = await FileUtils.uniqueFile(saveDir, fileName);
    tempFile = File('${targetFile.path}.part');
    await tempFile.writeAsBytes(bytes, flush: true);
    await tempFile.rename(targetFile.path);
    tempFile = null;

    var savedPath = targetFile.path;
    String? savedUri;
    if (Platform.isAndroid) {
      final saved = await PlatformIntegrations.saveToDownloads(
        path: targetFile.path,
        fileName: FileUtils.fileName(targetFile.path),
        mimeType: 'text/plain',
      );
      if (saved != null) {
        savedPath = saved.path;
        savedUri = saved.uri;
        try {
          await targetFile.delete();
        } catch (e) {
          debugPrint('[Server] Could not remove text staging file: $e');
        }
      }
    }

    var copied = false;
    try {
      copied = await (copyTextOverride ??
          PlatformIntegrations.copyTextToClipboard)(text);
    } catch (e) {
      debugPrint('[Server] Could not copy text to clipboard: $e');
    }

    final isDesktop = Platform.isLinux || Platform.isWindows;
    if (isDesktop && notifications) {
      await PlatformIntegrations.showDesktopNotification(
        'QuickDrop • Text received',
        '${TextTransfer.notificationPreview(text)} from $senderName'
            '${copied ? ' • copied to clipboard' : ' • saved without clipboard'}',
      );
    }

    final item = TransferItem(
      id: 'rx_${DateTime.now().microsecondsSinceEpoch}',
      fileName: FileUtils.fileName(savedPath),
      fileSize: totalBytes,
      filePath: savedPath,
      contentUri: savedUri,
      senderName: senderName,
      isIncoming: true,
      timestamp: DateTime.now(),
      success: true,
      kind: TransferKind.text,
      preview: TextTransfer.preview(text),
    );
    addIncomingHistory(item);

    request.response
      ..headers.contentType = ContentType.json
      ..statusCode = HttpStatus.ok
      ..write(
        jsonEncode({
          'status': 'success',
          'kind': 'text',
          'file': item.fileName,
          'path': item.filePath,
          'uri': item.contentUri,
          'size': item.fileSize,
          'clipboard': copied,
        }),
      );
    await request.response.close();
  } catch (e, st) {
    debugPrint('[Server] Error handling text transfer: $e\n$st');
    try {
      if (tempFile != null && await tempFile.exists()) {
        await tempFile.delete();
      }
    } catch (_) {}
    await _writeJsonError(
      request,
      HttpStatus.internalServerError,
      e.toString(),
    );
  }
}

Future<void> _writeJsonError(
  HttpRequest request,
  int status,
  String message,
) async {
  try {
    request.response
      ..headers.contentType = ContentType.json
      ..statusCode = status
      ..write(jsonEncode({'error': message}));
    await request.response.close();
  } catch (_) {}
}
```

```dart
static Future<TransferResult> sendText({
  required DevicePeer targetPeer,
  required String text,
  required String senderName,
}) async {
  final startTime = DateTime.now();
  final validation = TextTransfer.validate(text);
  if (validation != null) {
    return TransferResult(
      success: false,
      message: validation,
      durationMs: 0,
    );
  }

  final bytes = utf8.encode(text);
  final client = HttpClient();
  client.connectionTimeout = const Duration(seconds: 5);
  try {
    final request = await client
        .postUrl(
          Uri.parse(
            'http://${targetPeer.ip}:${targetPeer.httpPort}/api/transfer',
          ),
        )
        .timeout(const Duration(seconds: 10));

    request.headers.set(
      'X-File-Name',
      Uri.encodeComponent(TextTransfer.fileName()),
    );
    request.headers.set(
      'X-Sender-Name',
      Uri.encodeComponent(senderName),
    );
    request.headers.set(TextTransfer.header, '1');
    request.headers.set('X-File-Type', 'text/plain');
    request.headers.set(
      'Content-Type',
      'text/plain; charset=utf-8',
    );
    request.contentLength = bytes.length;
    request.add(bytes);

    final response = await request
        .close()
        .timeout(const Duration(seconds: 10));
    final responseBody = await utf8.decodeStream(response);
    final duration = DateTime.now().difference(startTime).inMilliseconds;

    if (response.statusCode == HttpStatus.ok) {
      String? savedPath;
      try {
        final data = jsonDecode(responseBody) as Map<String, dynamic>;
        savedPath = data['path'] as String?;
      } catch (_) {}
      return TransferResult(
        success: true,
        message: 'Text delivered in '
            '${FileUtils.formatDuration(Duration(milliseconds: duration))}',
        durationMs: duration,
        savedPath: savedPath,
        bytesSent: bytes.length,
      );
    }

    return TransferResult(
      success: false,
      message: response.statusCode == HttpStatus.requestEntityTooLarge
          ? 'The receiver rejected the text as too large.'
          : 'Receiver returned HTTP ${response.statusCode}',
      durationMs: duration,
      bytesSent: bytes.length,
    );
  } catch (e) {
    if (e.toString().contains('TimeoutException')) {
      return TransferResult(
        success: false,
        message: 'The receiver did not respond within 10 seconds.',
        durationMs: DateTime.now().difference(startTime).inMilliseconds,
      );
    }
    return TransferResult(
      success: false,
      message: linkDropMessage(e, targetPeer.link) ?? _friendlyError(e),
      durationMs: DateTime.now().difference(startTime).inMilliseconds,
    );
  } finally {
    client.close(force: true);
  }
}
```
